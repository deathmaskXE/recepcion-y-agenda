rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    match /equipos/{folio} {
      allow read, write: if request.auth != null;
    }

    match /estados_publicos/{folio} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /citas/{folio} {
      allow read, write: if request.auth != null;
    }

    match /citas_publicas/{folio} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /solicitudes_citas/{folio} {
      allow read, update, delete: if request.auth != null;
      allow create: if request.auth == null
        && request.resource.data.keys().hasOnly([
          'cliente', 'telefono', 'equipo', 'modelo', 'falla',
          'fecha', 'hora', 'estado', 'creada', 'actualizada'
        ])
        && request.resource.data.cliente is string
        && request.resource.data.cliente.size() >= 2
        && request.resource.data.cliente.size() <= 80
        && request.resource.data.telefono is string
        && request.resource.data.telefono.size() >= 10
        && request.resource.data.telefono.size() <= 20
        && request.resource.data.equipo is string
        && request.resource.data.equipo.size() >= 2
        && request.resource.data.equipo.size() <= 120
        && request.resource.data.fecha is string
        && request.resource.data.fecha.size() == 10
        && request.resource.data.hora is string
        && request.resource.data.hora.matches('^(10|11|12|13|14|15|16|17):00$')
        && request.resource.data.estado == 'Pendiente de confirmación';
    }

    match /disponibilidad_citas/{bloque} {
      allow read: if true;
      allow update, delete: if request.auth != null;
      allow create: if request.auth == null
        && request.resource.data.keys().hasOnly(['fecha', 'hora', 'solicitud'])
        && request.resource.data.fecha is string
        && request.resource.data.fecha.size() == 10
        && request.resource.data.hora is string
        && request.resource.data.hora.matches('^(10|11|12|13|14|15|16|17):00$')
        && request.resource.data.solicitud is string
        && request.resource.data.solicitud.matches('^XE-SOL-[0-9]{8}$');
    }
  }
}
